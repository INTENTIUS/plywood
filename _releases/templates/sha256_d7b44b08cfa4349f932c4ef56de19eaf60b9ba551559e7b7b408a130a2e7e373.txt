name: plywood

on:
  push:
    branches:
      - main
  pull_request: {}

permissions:
  contents: read

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: '${{ github.ref != ''refs/heads/main'' }}'

jobs:
  pkg:
    name: package
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - name: Setup Node.js
        uses: actions/setup-node@949feb2413d6458794dcd2491c4babbbce0c15c1
        with:
          node-version: '24'
          cache: npm
          cache-dependency-path: package-lock.json
      - name: Install
        run: npm ci
      - name: Typecheck
        run: npm run typecheck
      - name: Build
        run: npm run build
      - name: Unit tests
        run: npm test
      - name: Bundle check
        run: npm run bundle-check -w @intentius/plywood
      - name: Generated files match their sources
        run: npm run generated-check
  site:
    name: site
    runs-on: ubuntu-latest
    timeout-minutes: 10
    defaults:
      run:
        working-directory: site
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - name: Setup Node.js
        uses: actions/setup-node@949feb2413d6458794dcd2491c4babbbce0c15c1
        with:
          node-version: '24'
          cache: npm
          cache-dependency-path: site/package-lock.json
      - name: Install
        run: npm ci
      - name: Lint prose
        run: npm run lint:prose
      - name: Build the site
        run: npm run build
